def compare_cafe_restaurant_counts(intent_target, previous_observations=None, intent_text=None, tag_a="amenity=cafe", tag_b="amenity=restaurant"):
    """実行時に指定された対象の前段 Observation から amenity=cafe と amenity=restaurant の count を比較し、winner を返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    candidates = []
    for value in decoded:
        candidates += value if isinstance(value, list) else [value]
    target = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    assert target

    by_tag = {o["tag"]: o for o in records}
    assert tag_a in by_tag and tag_b in by_tag
    a, b = by_tag[tag_a], by_tag[tag_b]
    winner = tag_a if a["count"] >= b["count"] else tag_b
    return {
        "name": target["name"],
        "relation_id": target["relation_id"],
        "results": [
            {"tag": tag_a, "count": a["count"]},
            {"tag": tag_b, "count": b["count"]},
        ],
        "winner": winner,

## find_max_latitude_stations@v1
from geo_voyager.control_primitives import connect_duckdb, load_stations


def find_max_latitude_stations(dataset_id):
    """駅データから緯度が最大の駅を特定し、駅名と緯度を返す。同緯度で最大が複数ある場合はすべて返す。"""
    with connect_duckdb() as connection:
        # まず最大緯度を取得
        max_lat_row = load_stations(dataset_id, connection).order("latitude DESC").limit(1).fetchone()
        if max_lat_row is None:
            raise ValueError("No station found")
        max_lat = max_lat_row[1]
        
        # その緯度を持つすべての駅を取得
        # DuckDB relation のフィルタリングは .filter() または SQL 的なアプローチが必要だが、
        # 提供された Primitive の説明に filter は明示されていないが、
        # load_stations が relation を返すため、標準的な SQL 操作が可能か確認。
        # 説明に "order", "limit", "aggregate" はあるが filter は無いため、
        # 一度取得して Python 側でフィルタリングするのが安全。
        # ただし、データセットが大きすぎる可能性があるため、
        # 可能な限り DB 側で処理したいが、filter メソッドの存在が不明。
        # 代わりに、load_stations を使い直して、最大緯度と一致する行を探す。
        # Python 側でフィルタリングする場合、全データを取得する必要がある。
        # 駅データは比較的少ないため、全取得 -> フィルタ が現実的。
        stations = load_stations(dataset_id, connection).fetchall()
        
    results = []
    for station in stations:
        # station は (name, latitude, longitude) の tuple
        name, latitude, longitude = station
        if latitude == max_lat:
            results.append({"name": name, "latitude": latitude, "longitude": longitude})
            
    if not results:
        raise ValueError("No stations with max latitude found")
        
    return results

## get_relation_id@v1 (seed)
import json

from geo_voyager.control_primitives import call_service


def get_relation_id(intent_target):
    """Find the OSM relation of a named place (for example a ward) with Nominatim and return its name and relation_id. When the target already carries a relation_id, that id is returned without a search."""
    if intent_target.get("id_type") == "relation_id" and intent_target.get("id_value"):
        return {"name": intent_target["name"], "relation_id": str(intent_target["id_value"])}
    results = json.loads(call_service("nominatim", path="/search",
                                      params={"q": intent_target["name"], "format": "jsonv2", "limit": "10"}))
    relations = [item for item in results if item.get("osm_type") == "relation"]
    if not relations:
        raise ValueError(f"No OSM relation found for {intent_target['name']}")
    return {"name": intent_target["name"], "relation_id": str(relations[0]["osm_id"])}

import json
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def get_wards_population_summary(dataset_id, previous_observations, intent_text=None):
    """前段の区名リストに基づき、各区の人口と合計人口を取得して返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    target_names = []
    for obj in decoded:
        if isinstance(obj, list):
            for item in obj:
                if isinstance(item, dict) and "name" in item:
                    target_names.append(str(item["name"]))
    
    if not target_names:
        raise ValueError("No ward names found in previous observations")
    
    # 重複除去と順序維持（必要に応じて）
    target_names = list(dict.fromkeys(target_names))
    
    with connect_duckdb() as connection:
        relation = load_admin_units(dataset_id, connection, area="東京都23区")
        rows = relation.fetchall()
        
        # 名前と人口の辞書を作成
        name_to_population = {str(row[1]): row[2] for row in rows}
        
        wards_data = []
        total_population = 0
        
        for name in target_names:
            if name in name_to_population:
                population = name_to_population[name]
                wards_data.append({"name": name, "population": population})
                total_population += population
            else:
                raise ValueError(f"Ward '{name}' not found in dataset")
        
        # 前段の名前順ではなく、通常はコード順や人口順などあるが、Intentでは「一覧で出力」とだけあるため、
        # 前段の取得順（または辞書の順）を維持するか、code5でソートしてもよい。
        # ここでは前段のリスト順を維持するが、コード順が自然かもしれない。
        # 簡潔さのため、前段の順を維持する。
        
        return {
            "wards": wards_data,
            "total_population": total_population
        }

def list_all_ward_relation_ids():
    """YuisekinGeoSPARQL の gs:Ward 全23個体の日本語ラベルと OSM relation ID を取得して返す。"""
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> "
        "SELECT ?label ?relation WHERE { "
        "?ward a gs:Ward ; rdfs:label ?label ; gs:osmRelation ?relation . "
        "FILTER(LANG(?label) = \"ja\") . "
        "} ORDER BY ?label"
    )
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    payload = json.loads(response)
    rows = payload["results"]["bindings"]
    assert len(rows) == 23
    result = []
    for row in rows:
        relation_id = row["relation"]["value"].rsplit("/", 1)[-1]
        assert str(relation_id).isdigit()
        result.append({"name": row["label"]["value"], "relation_id": relation_id})
    return result

## summarize_ramen_sushi_counts@v1
import json


def summarize_ramen_sushi_counts(previous_observations, intent_text=None, tag_a="cuisine=ramen", tag_b="cuisine=sushi"):
    """前段 Observation から tag=cuisine=ramen と tag=cuisine=sushi の count を読み、両方を results 配列の tag/count として、多い方の winner_tag と winner_count を含めて JSON で出力する。"""
    counts = {}
    for observation in previous_observations:
        item = json.loads(observation)
        tag = item["tag"]
        count = item["count"]
        counts[tag] = count

    count_a = counts[tag_a]
    count_b = counts[tag_b]

    if count_a >= count_b:
        winner_tag = tag_a
        winner_count = count_a
    else:
        winner_tag = tag_b
        winner_count = count_b

    return {
        "results": [
            {"tag": tag_a, "count": count_a},
            {"tag": tag_b, "count": count_b},
        ],
        "winner_tag": winner_tag,
        "winner_count": winner_count,
    }

from geo_voyager.control_primitives import call_service


def tag_usage_count(key, value):
    """Count how many times the OSM tag key=value is used in the whole OSM database, with Taginfo's tag statistics. Returns the tag and the count."""
    payload = json.loads(call_service("taginfo", path="/api/4/tag/stats", params={"key": key, "value": value}))
    total = [item["count"] for item in payload["data"] if item["type"] == "all"]
    if len(total) != 1:
        raise ValueError(f"Taginfo returned no total for {key}={value}")
    return {"key": key, "value": value, "count": total[0]}

## total_population@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
    with connect_duckdb() as connection:
        total = load_admin_units(dataset_id, connection, area=area).aggregate("sum(population) AS population_total").fetchone()[0]
    if type(total) is not int:
        raise TypeError("The population total must be an integer")
    return total
