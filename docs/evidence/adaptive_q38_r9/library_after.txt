## compare_two_counts@v1
import json


def compare_two_counts(intent_target, previous_observations, intent_text, tag_a="amenity=cafe", tag_b="amenity=restaurant"):
    """実行時に指定された対象の2つのタグ件数を比較し、大きい方を返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    candidates = []
    for value in decoded:
        candidates += value if isinstance(value, list) else [value]
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    assert matches
    target = matches[0]
    records = [o for o in candidates if isinstance(o, dict) and str(o.get(intent_target["id_type"])) == intent_target["id_value"] and o.get("tag") in (tag_a, tag_b) and "count" in o]
    assert len(records) == 2
    a = next(o for o in records if o["tag"] == tag_a)
    b = next(o for o in records if o["tag"] == tag_b)
    count_a = int(a["count"]["count"])
    count_b = int(b["count"]["count"])
    larger = tag_a if count_a > count_b else tag_b
    return {
        "name": target["name"],
        "relation_id": target["relation_id"],
        "tag": f"{tag_a} vs {tag_b}",
        "count": {tag_a: count_a, tag_b: count_b},
        "larger": larger
    }

## compare_ward_counts@v1
import json


def compare_ward_counts(previous_observations, intent_text, tag="amenity=cafe"):
    """前段の Observation から指定 tag の count を持つ測定レコードを抽出し、両区の count とより多い区を winner として返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    records = [o for o in decoded if o.get("tag") == tag and "count" in o]
    assert len(records) == 2
    result = {}
    for record in records:
        name = record["name"]
        relation_id = record["relation_id"]
        count = record["count"]["count"]
        assert isinstance(count, int)
        result[name] = {"name": name, "relation_id": relation_id, "count": count}
    assert len(result) == 2
    winner = max(result.items(), key=lambda item: item[1]["count"])[0]
    return {
        "wards": [result[name] for name in result],
        "winner": winner
    }

    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})

## get_relation_id@v1 (seed)
import json

from geo_voyager.control_primitives import call_service


def get_relation_id(intent_target):
    """Find the OSM relation of a named place (for example a ward) with Nominatim and return its name and relation_id. When the target already carries a relation_id, that id is returned without a search."""
    if intent_target.get("id_type") == "relation_id" and intent_target.get("id_value"):
        return {"name": intent_target["name"], "relation_id": str(intent_target["id_value"])}
    results = json.loads(call_service("nominatim", path="/search",
                                      params={"q": intent_target["name"], "format": "jsonv2", "limit": "10"}))
    relations = [item for item in results if item.get("osm_type") == "relation"]
    if not relations:
        raise ValueError(f"No OSM relation found for {intent_target['name']}")
    return {"name": intent_target["name"], "relation_id": str(relations[0]["osm_id"])}

## get_ward_population_by_name@v1
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def get_ward_population_by_name(dataset_id, intent_target=None, area="東京都23区"):
    """
    指定された対象区域（意図ターゲット）の名称に基づき、
    Dataset からその区域の人口を特定して name と population を持つ dict を返す。
    """
    # intent_target のデフォルトは「世田谷区」だが、汎用性を高めるため引数から取得
    if intent_target is None:
        target_name = "世田谷区"
    else:
        # intent_target は通常 {"name": "..."} のような dict または str である可能性があり、
        # ここでは name キーがある dict を想定して安全に取得する。
        if isinstance(intent_target, dict) and "name" in intent_target:
            target_name = intent_target["name"]
        elif isinstance(intent_target, str):
            target_name = intent_target
        else:
            # フォールバック: 既定の対象
            target_name = "世田谷区"

    with connect_duckdb() as connection:
        # 東京都23区の行政区域情報をロード
        # 列: code5, name, population
        relation = load_admin_units(dataset_id, connection, area=area)
        
        # 対象の名称でフィルタリング
        # DuckDB relation の操作は SQL-like なメソッドか、fetchall で全取得して Python 側でフィルタ
        # ここで fetchone などは特定の行を直接取得できないため（WHERE 構文がないため）、
        # 全行を取得して Python 側でマッチングするか、relation を SQL 化できない場合があるため、
        # 安全策として fetchall() を使用し、name が一致する行を探す。
        rows = relation.fetchall()
        
        result_row = None
        for row in rows:
            # row の構造は (code5, name, population)
            row_name = row[1]
            if row_name == target_name:
                result_row = row
                break
        
        if result_row is None:
            raise ValueError(f"Target area '{target_name}' not found in dataset area '{area}'")
        
        name = result_row[1]
        population = result_row[2]
        
        return {"name": name, "population": population}

## list_all_ward_names_and_relation_ids@v1
import json
from geo_voyager.control_primitives import call_service


def list_all_ward_names_and_relation_ids(expected_count=23):
    """YuisekinGeoSPARQLの全gs:Wardの日本語labelとosmRelation末尾数値をnameとrelation_idの一覧として返す。"""
    count_result = count_gs_ward_instances()
    if count_result["count"] != expected_count:
        raise ValueError(f"Expected {expected_count} gs:Ward, got {count_result['count']}")
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> "
        "SELECT ?label ?relation WHERE { "
        "?ward a gs:Ward ; rdfs:label ?label ; gs:osmRelation ?relation . "
        'FILTER(LANG(?label) = "ja") } ORDER BY ?label'
    )
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    payload = json.loads(response)
    rows = payload["results"]["bindings"]
    if len(rows) != expected_count:
        raise ValueError(f"Expected {expected_count} rows, got {len(rows)}")
    wards = []
    for row in rows:
        name = row["label"]["value"]
        relation_id = row["relation"]["value"].rsplit("/", 1)[-1]
        assert str(relation_id).isdigit()
        wards.append({"name": name, "relation_id": relation_id})
    return wards

## sum_23_ward_population@v1
import json


def sum_23_ward_population(previous_observations, intent_text=None):
    """前段の東京23区全23区の population を合計し、total_population を整数で返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    records = [
        item for item in decoded
        if isinstance(item, list) and len(item) == 23 and all(
            isinstance(record, dict) and "name" in record and "relation_id" in record and "population" in record
            for record in item
        )
    ]
    if len(records) != 1:
        raise ValueError(f"23 ward population records are required, {len(records)} found")
    records = records[0]
    if len({(item["name"], item["relation_id"]) for item in records}) != 23:
        raise ValueError("Ward population records must be unique by name and relation_id")
    total = 0
    for item in records:
        population = item["population"]
        if type(population) is not int:
            raise TypeError("population must be an integer")
        total += population
    return {"total_population": total}

import json

from geo_voyager.control_primitives import call_service


def tag_usage_count(key, value):
    """Count how many times the OSM tag key=value is used in the whole OSM database, with Taginfo's tag statistics. Returns the tag and the count."""
    payload = json.loads(call_service("taginfo", path="/api/4/tag/stats", params={"key": key, "value": value}))
    total = [item["count"] for item in payload["data"] if item["type"] == "all"]
    if len(total) != 1:
        raise ValueError(f"Taginfo returned no total for {key}={value}")
    return {"key": key, "value": value, "count": total[0]}

from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def top_areas_by_population(dataset_id, count, area=None):
    """List the given number of administrative areas with the largest populations, largest first, in all areas or in one named set (for example 東京都23区). Each item has its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        rows = load_admin_units(dataset_id, connection, area=area).order("population DESC, code5 ASC").limit(count).fetchall()
    if len(rows) != count:
        raise ValueError(f"{count} areas are required, {len(rows)} found")
    if any(type(population) is not int for _, _, population in rows):
        raise TypeError("population must be an integer")
    return [{"code5": code5, "name": name, "population": population} for code5, name, population in rows]

import json
from geo_voyager.control_primitives import call_service


def top_key_values_by_count(key="cuisine", limit=3):
    """taginfo の /api/4/key/values を用いて指定キーの値の使用数を取得し、使用数の多い順に上位件数の値と使用数を返す。"""
    payload = json.loads(call_service("taginfo", path="/api/4/key/values", params={"key": key, "page": "1", "rp": str(limit), "sortname": "count", "sortorder": "desc"}))
    data = payload.get("data")
    if not isinstance(data, list) or len(data) < limit:
        raise ValueError(f"taginfo returned insufficient data for key={key}")
    return [{"value": item["value"], "count": item["count"]} for item in data[:limit]]

## total_population@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
