from geo_voyager.control_primitives import call_service
import json


def calculate_pedestrian_duration(start_lat=35.6580, start_lon=139.7016, end_lat=35.6896, end_lon=139.7006):
    """Calculate walking duration in seconds between two coordinates."""
    body = json.dumps({
        "locations": [
            {"lat": start_lat, "lon": start_lon},
            {"lat": end_lat, "lon": end_lon}
        ],
        "costing": "pedestrian",
        "units": "kilometers"
    })
    result = call_service("valhalla", path="/route", body=body, content_type="application/json")
    data = json.loads(result)
    trip = data.get("trip", {})
    summary = trip.get("summary", {})
    duration = summary.get("time")
    if duration is None:
        raise ValueError("No duration found in route response")
    return {"duration_seconds": duration}

from geo_voyager.control_primitives import call_service


def compare_cafe_and_restaurant(intent_target, previous_observations=None, intent_text=None):
    """Count and compare amenity=cafe and amenity=restaurant in the runtime target area."""
    if previous_observations:
        import json
        candidates = []
        for text in previous_observations:
            value = json.loads(text)
            candidates += value if isinstance(value, list) else [value]
        matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
        assert matches, "Target not found in previous observations"
        target = matches[0]
        relation_id = target["relation_id"]
    else:
        raise ValueError("Previous observations required")

    area_id = int(relation_id) + 3600000000
    
    ql_cafe = f"[out:json][timeout:12];nwr[\"amenity\"=\"cafe\"](area:{area_id});out count;"
    result_cafe = call_service("overpass", path="/api/interpreter", body=ql_cafe, content_type="text/plain")
    data_cafe = json.loads(result_cafe)
    if not data_cafe.get("elements"):
        raise ValueError("No cafe count result")
    cafe_count = int(data_cafe["elements"][0]["tags"]["total"])
    
    ql_rest = f"[out:json][timeout:12];nwr[\"amenity\"=\"restaurant\"](area:{area_id});out count;"
    result_rest = call_service("overpass", path="/api/interpreter", body=ql_rest, content_type="text/plain")

    if not data_rest.get("elements"):
        raise ValueError("No restaurant count result")

        "name": intent_target["name"],
        "relation_id": relation_id,
        "cafe_count": cafe_count,
        "restaurant_count": restaurant_count,
        "more_common": more
    }

## convert_seconds_to_minutes@v1
import json


def convert_seconds_to_minutes(previous_observations=None, intent_text="2904.379秒を分単位に換算した値"):
    """Convert pedestrian duration from seconds to minutes."""
    if not previous_observations:
        raise ValueError("previous_observations is required")
    
    # Parse the previous observation to get duration_seconds
    # Assuming the first observation contains the duration data
    try:
        obs = json.loads(previous_observations[0])
        duration_seconds = obs["duration_seconds"]
    except (IndexError, json.JSONDecodeError, KeyError) as e:
        raise ValueError(f"Failed to parse previous observation: {e}")
    
    # Convert seconds to minutes
    minutes = duration_seconds / 60.0
    
    return {"minutes": minutes}

## count_wards_and_get_ids@v1
from geo_voyager.control_primitives import call_service
import json


def count_wards_and_get_ids(intent_target, previous_observations=None, intent_text=None):
    """Count gs:Ward instances and return their Japanese labels and OSM relation IDs."""
    query = """
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    SELECT ?label ?relation WHERE {
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      ?ward gs:osmRelation ?relation .
    }
    ORDER BY ?label
    """
    params = {"query": query}
    result = call_service("yuisekin-geosparql", path="/geo/sparql", params=params)
    data = json.loads(result)
    
    wards = []
    for row in data["results"]["bindings"]:
        label = row["label"]["value"]
        relation_uri = row["relation"]["value"]
        # Extract numeric ID from URI like http://example.com/relation/123
        relation_id_str = relation_uri.rsplit("/", 1)[-1]
        assert str(relation_id_str).isdigit(), f"Invalid relation ID: {relation_id_str}"
        wards.append({"name": label, "relation_id": int(relation_id_str)})
        
    return {"count": len(wards), "wards": wards}

import json


def count_wards_with_ku_suffix(intent_target=None, previous_observations=None, intent_text=None):
    """Count wards whose Japanese name is exactly 4 characters and ends with '区'."""
    if not previous_observations:
        raise ValueError("Previous observations required")
    
    decoded = [json.loads(text) for text in previous_observations]
    
    wards = None
    for obs in decoded:
        if isinstance(obs, dict) and "wards" in obs:
            wards = obs["wards"]
            break
            
    if wards is None:
        raise ValueError("No wards data found in previous observations")
        
    assert len(wards) == 23, f"Expected 23 wards, got {len(wards)}"
    
    count = 0
    for ward in wards:
        label = ward.get("label", "")
        if len(label) == 4 and label.endswith("区"):
            count += 1
            
    return {"count": count}

    if not target_name:
        raise ValueError("Target name is required")
    query = f"""
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    SELECT ?label ?relation
    WHERE {{
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      ?ward gs:osmRelation ?relation .
      FILTER(STR(?label) = "{target_name}")
    }}
    ORDER BY ?label
    """
    result = call_service("yuisekin-geosparql", path="/geo/sparql", body=query, content_type="application/sparql-query")
    data = __import__("json").loads(result)
    bindings = data.get("results", {}).get("bindings", [])
    if not bindings:
        raise ValueError("No ward found")
    row = bindings[0]
    relation_uri = row["relation"]["value"]
    relation_id = relation_uri.rsplit("/", 1)[-1]
    assert str(relation_id).isdigit(), f"Invalid relation ID: {relation_id}"

from geo_voyager.control_primitives import call_service, load_admin_units
import json


def get_south_boundary(intent_target, name=None):
    """Get the southern latitude of the bounding box for the runtime target area."""
    target_name = intent_target.get("name") if isinstance(intent_target, dict) else name
    if not target_name:
        raise ValueError("Target name is required")
    
    # Use load_admin_units to get the area info directly if available, 
    # otherwise fallback to nominatim search.
    # Since dataset_id is not provided in the function signature, we assume 
    # the environment provides it or we rely on the service. 
    # However, the previous code used nominatim. Let's stick to nominatim but fix the query.
    # The error "No result found for relation" suggests the query "relation(XXX)" 
    # didn't return results or the structure was different.
    # Nominatim search for "relation(XXX)" is not standard. 
    # Usually, we search by name or use /reverse with coordinates.
    # But we need the bounding box. 
    # Let's try searching for the area name directly.
    
    params = {
        "format": "jsonv2",
        "q": target_name,
        "limit": "10"
    }
    result = call_service("nominatim", path="/search", params=params)
    data = json.loads(result)
    
    if not data:
        raise ValueError("No result found for target")
    
    # Find the best match, preferably an administrative boundary
    best_match = None
    for item in data:
        if item.get('class') == 'boundary' and item.get('type') in ['administrative', 'boundary']:
            best_match = item
            break
    
    if best_match is None:
        # If no administrative boundary found, use the first result
        best_match = data[0]
        
    bbox = best_match.get("boundingbox")
    if bbox is None:
        raise ValueError("No bounding box found")
    
    south = float(bbox[0])
    return {"name": target_name, "south_latitude": south}

    for row in bindings:
        label = row["label"]["value"]
        relation_uri = row["relation"]["value"]
        
        # Extract numeric ID from the end of the URI

        # Assert that the extracted ID is a valid digit string

        wards_list.append({

def top_areas_by_population(dataset_id, count, area=None):
    """List the given number of administrative areas with the largest populations, largest first, in all areas or in one named set (for example 東京都23区). Each item has its 5-digit code, name and population."""
    with connect_duckdb() as connection:

## total_population@v1
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
    with connect_duckdb() as connection:
        total = load_admin_units(dataset_id, connection, area=area).aggregate("sum(population) AS population_total").fetchone()[0]
    if type(total) is not int:
        raise TypeError("The population total must be an integer")
    return total
