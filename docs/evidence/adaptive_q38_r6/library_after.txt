## check_area_exists_for_target@v1
import json
from geo_voyager.control_primitives import call_service


def check_area_exists_for_target(intent_target, previous_observations, intent_text):
    """実行時に指定された対象の OSM relation ID を Overpass の area ID に変換し、対応する area 要素の存在を確認して、対象の name と relation_id、変換後の area_id、area_exists を返す。"""
    candidates = []
    for text in previous_observations:
        value = json.loads(text)
        candidates += value if isinstance(value, list) else [value]
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    assert matches
    target = matches[0]
    relation_id = str(target["relation_id"])
    area_id = int(relation_id) + 3600000000
    query = f"[out:json][timeout:12];area({area_id});out ids;"
    response = call_service("overpass", path="/api/interpreter", params={"data": query})
    payload = json.loads(response)
    if payload.get("remark"):
        raise RuntimeError(payload["remark"])
    area_exists = any(el.get("type") == "area" and el.get("id") == area_id for el in payload.get("elements", []))
    if not area_exists:
        raise ValueError(f"area {area_id} not found")
    return {"name": target["name"], "relation_id": relation_id, "area_id": area_id, "area_exists": area_exists}

def compare_cafe_and_restaurant_counts(intent_target, previous_observations, intent_text, cafe_tag="amenity=cafe", restaurant_tag="amenity=restaurant"):
    """実行時に指定された対象の前段 Observation から cafe と restaurant の件数を比較して返す。"""
    candidates = []
    for text in previous_observations:
        value = json.loads(text)
        candidates += value if isinstance(value, list) else [value]
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    assert matches
    target = matches[0]
    cafe = [o for o in candidates if isinstance(o, dict) and str(o.get(intent_target["id_type"])) == intent_target["id_value"] and o.get("tag") == cafe_tag and "count" in o]
    restaurant = [o for o in candidates if isinstance(o, dict) and str(o.get(intent_target["id_type"])) == intent_target["id_value"] and o.get("tag") == restaurant_tag and "count" in o]

    return {

        "relation_id": target["relation_id"],

        "restaurant_count": restaurant_count,

## get_relation_id@v1
import json

from geo_voyager.control_primitives import call_service


def get_relation_id(intent_target):
    """Find the OSM relation of a named place (for example a ward) with Nominatim and return its name and relation_id. When the target already carries a relation_id, that id is returned without a search."""
    if intent_target.get("id_type") == "relation_id" and intent_target.get("id_value"):
        return {"name": intent_target["name"], "relation_id": str(intent_target["id_value"])}
    results = json.loads(call_service("nominatim", path="/search",
                                      params={"q": intent_target["name"], "format": "jsonv2", "limit": "10"}))
    relations = [item for item in results if item.get("osm_type") == "relation"]
    if not relations:
        raise ValueError(f"No OSM relation found for {intent_target['name']}")
    return {"name": intent_target["name"], "relation_id": str(relations[0]["osm_id"])}

## get_route_legs_summary@v1
import json
from geo_voyager.control_primitives import call_service


def get_route_legs_summary(origin_lat=35.6580, origin_lon=139.7016, destination_lat=35.6896, destination_lon=139.7006, costing="pedestrian"):
    """出発点と到着点の座標を引数に受け取り、指定した costing で Valhalla の /route を呼び、costing、summary の length と time、各 leg の summary.length と summary.time を欠落なく一覧で出力し、各 leg の time の合計と summary.time の一致、および summary.time を分単位に変換した値を算出する。"""
    summary = route_summary(origin_lat, origin_lon, destination_lat, destination_lon, costing=costing)
    request = {"locations": [{"lat": origin_lat, "lon": origin_lon}, {"lat": destination_lat, "lon": destination_lon}], "costing": costing, "units": "kilometers"}
    payload = json.loads(call_service("valhalla", path="/route", body=json.dumps(request), content_type="application/json"))
    legs = payload["trip"]["legs"]
    if not legs:
        raise ValueError("No legs in route response")
    legs_out = []
    for leg in legs:
        if "summary" not in leg or "length" not in leg["summary"] or "time" not in leg["summary"]:
            raise ValueError("Missing leg summary fields")
        legs_out.append({"length_km": leg["summary"]["length"], "time_s": leg["summary"]["time"]})
    leg_time_total = sum(leg["time_s"] for leg in legs_out)
    return {"costing": costing, "summary_length_km": summary["length_km"], "summary_time_s": summary["time_s"], "legs": legs_out, "leg_time_total_s": leg_time_total, "leg_time_matches_summary": leg_time_total == summary["time_s"], "summary_time_min": summary["time_min"]}

def least_populous_area(dataset_id, area=None):
    """Find the administrative area with the smallest population, in all areas or in one named set (for example 東京都23区). Returns its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        row = load_admin_units(dataset_id, connection, area=area).order("population ASC, code5 ASC").limit(1).fetchone()

        raise ValueError("No administrative area found")

        '} ORDER BY ?label'
    )
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    payload = json.loads(response)
    bindings = payload["results"]["bindings"]
    assert len(bindings) == 23, f"Expected 23 gs:Ward rows, got {len(bindings)}"
    wards = []
    for row in bindings:
        label = row["label"]["value"]
        relation = row["relation"]["value"]
        relation_id = relation.rsplit("/", 1)[-1]
        assert str(relation_id).isdigit()
        wards.append({"name": label, "relation_id": relation_id})
    return wards

## min_relation_id_ward@v1
import json


def min_relation_id_ward(previous_observations, intent_text):
    """前段 Observation の東京23区全23件の name と relation_id を解析し、relation_id の数値が最小の1件を name と relation_id で返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    candidates = []
    for value in decoded:
        if isinstance(value, list):
            candidates.extend(value)
        elif isinstance(value, dict):
            candidates.append(value)
    wards = [
        item for item in candidates
        if isinstance(item, dict) and "name" in item and "relation_id" in item
    ]
    assert len(wards) == 23, f"Expected 23 wards, got {len(wards)}"
    best = min(wards, key=lambda item: int(item["relation_id"]))
    return {"name": best["name"], "relation_id": best["relation_id"]}

from geo_voyager.control_primitives import call_service


def route_summary(origin_lat, origin_lon, destination_lat, destination_lon, costing="auto"):
    """Route between two points with Valhalla and return the length in kilometres and the time in seconds and minutes. costing is auto, pedestrian or bicycle."""
    request = {"locations": [{"lat": origin_lat, "lon": origin_lon}, {"lat": destination_lat, "lon": destination_lon}],
               "costing": costing, "units": "kilometers"}
    summary = json.loads(call_service("valhalla", path="/route", body=json.dumps(request),

    return {"costing": costing, "length_km": summary["length"], "time_s": summary["time"], "time_min": summary["time"] / 60}

## tag_usage_count@v1
import json

from geo_voyager.control_primitives import call_service


def tag_usage_count(key, value):
    """Count how many times the OSM tag key=value is used in the whole OSM database, with Taginfo's tag statistics. Returns the tag and the count."""
    payload = json.loads(call_service("taginfo", path="/api/4/tag/stats", params={"key": key, "value": value}))
    total = [item["count"] for item in payload["data"] if item["type"] == "all"]
    if len(total) != 1:
        raise ValueError(f"Taginfo returned no total for {key}={value}")
    return {"key": key, "value": value, "count": total[0]}

def top_areas_by_population(dataset_id, count, area=None):
    """List the given number of administrative areas with the largest populations, largest first, in all areas or in one named set (for example 東京都23区). Each item has its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        rows = load_admin_units(dataset_id, connection, area=area).order("population DESC, code5 ASC").limit(count).fetchall()

def top_cuisine_values_by_usage(key="cuisine", limit=3):

## total_population@v1
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
    with connect_duckdb() as connection:
        total = load_admin_units(dataset_id, connection, area=area).aggregate("sum(population) AS population_total").fetchone()[0]
    if type(total) is not int:
        raise TypeError("The population total must be an integer")
    return total
