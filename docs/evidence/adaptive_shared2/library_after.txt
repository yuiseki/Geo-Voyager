## calculate_minutes_from_seconds@v1
from geo_voyager.control_primitives import call_service
import json


def calculate_minutes_from_seconds(intent_target=None, previous_observations=None, intent_text=""):
    """Calculate minutes by dividing seconds from previous observations by 60."""
    if not previous_observations:
        raise ValueError("No previous observations provided")
    
    # Find the observation containing time_seconds
    decoded = [json.loads(text) for text in previous_observations]
    time_obs = None
    for obs in decoded:
        if "time_seconds" in obs:
            time_obs = obs
            break
            
    if time_obs is None:
        raise ValueError("No time_seconds found in previous observations")
        
    time_seconds = time_obs["time_seconds"]
    minutes = time_seconds / 60.0
    
    return {"minutes": minutes}

## calculate_route_distance@v1
from geo_voyager.control_primitives import call_service
import json


def calculate_route_distance(start_lat=35.6580, start_lon=139.7016, end_lat=35.6896, end_lon=139.7006):
    """Calculate the driving distance in kilometers between two coordinates."""
    body = json.dumps({
        "locations": [
            {"lat": start_lat, "lon": start_lon},
            {"lat": end_lat, "lon": end_lon}
        ],
        "costing": "auto",
        "units": "kilometers"
    })
    response = call_service("valhalla", path="/route", body=body, content_type="application/json")
    data = json.loads(response)
    trip = data.get("trip")
    if not trip:
        raise ValueError("No trip found in response")
    length = trip.get("summary", {}).get("length")
    if length is None:
        raise ValueError("Length not found in trip summary")
    return {"distance_km": length}

import json
from geo_voyager.control_primitives import call_service


def compare_amenities_in_target(intent_target, previous_observations=None, intent_text="", cafe_key="amenity", cafe_value="cafe", restaurant_key="amenity", restaurant_value="restaurant"):
    """Compare counts of cafe and restaurant in the runtime target area."""
    if previous_observations is None:
        previous_observations = []
    
    results = []
    for text in previous_observations:
        if text:
            try:
                value_obj = json.loads(text)
                if isinstance(value_obj, list):
                    results.extend(value_obj)
                else:
                    results.append(value_obj)
            except json.JSONDecodeError:
                pass

    cafe_data = None
    restaurant_data = None
    
    for r in results:
        if isinstance(r, dict) and r.get("tag") == f"{cafe_key}={cafe_value}":
            cafe_data = r
        elif isinstance(r, dict) and r.get("tag") == f"{restaurant_key}={restaurant_value}":
            restaurant_data = r
            
    if cafe_data is None or restaurant_data is None:
        raise ValueError("Missing cafe or restaurant data in previous observations")
        
    cafe_count = cafe_data["count"]
    restaurant_count = restaurant_data["count"]
    
    if cafe_count >= restaurant_count:
        return {"name": cafe_data["name"], "relation_id": cafe_data["relation_id"], "tag": cafe_data["tag"], "count": cafe_count, "winner": "cafe"}
    else:
        return {"name": restaurant_data["name"], "relation_id": restaurant_data["relation_id"], "tag": restaurant_data["tag"], "count": restaurant_count, "winner": "restaurant"}

    if previous_observations is None:
        previous_observations = []
    
    candidates = []
    for text in previous_observations:
        if text:
            try:
                value_obj = json.loads(text)
                if isinstance(value_obj, list):
                    candidates.extend(value_obj)
                else:
                    candidates.append(value_obj)

                pass

        raise ValueError("Target not found in previous observations")

## filter_wards_by_name_length@v1
import json


def filter_wards_by_name_length(previous_observations, min_length=4, max_length=4):
    """Filter wards from previous observations by Japanese name length containing '区'."""
    if not previous_observations:
        raise ValueError("No previous observations provided")
    
    decoded = json.loads(previous_observations[0])
    
    if not isinstance(decoded, list):
        raise TypeError("Expected a list in previous observations")
    
    if len(decoded) != 23:
        raise ValueError(f"Expected 23 wards, got {len(decoded)}")
    
    count = 0
    for ward in decoded:
        name = ward.get("name")
        if name is None:
            raise KeyError("Ward object missing 'name' field")
        if "区" in name and min_length <= len(name) <= max_length:
            count += 1
            
    return {"count": count}

def get_setagaya_south_lat(intent_target, query="世田谷区"):

    params = {"q": query, "format": "jsonv2", "limit": "1"}
    response = call_service("nominatim", path="/search", params=params)
    data = json.loads(response)
    if not data:

    bbox = item.get("boundingbox")

    osm_id = item.get("osm_id")
    osm_type = item.get("osm_type")
    if osm_type == "relation":
        rel_id = osm_id
    elif osm_type == "way":
        rel_id = osm_id + 3600000000
    else:
        rel_id = osm_id
    return {"name": item.get("name"), "relation_id": rel_id, "south_lat": south_lat}

## get_shinjuku_relation_id@v1
from geo_voyager.control_primitives import call_service


def get_shinjuku_relation_id(intent_target, name=None):
    """Get the OSM relation ID for the runtime-specified ward."""
    target_name = name if name else intent_target.get("name")
    if not target_name:
        raise ValueError("Target name is required")
    
    query = f"""
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    
    SELECT ?label ?relation
    WHERE {{
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      FILTER(STR(?label) = "{target_name}")
      ?ward gs:osmRelation ?relation .
    }}
    ORDER BY ?label
    """
    
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    import json
    data = json.loads(response)
    
    bindings = data.get("results", {}).get("bindings", [])
    if not bindings:
        raise ValueError(f"No ward found with name {target_name}")
    
    row = bindings[0]
    relation_uri = row["relation"]["value"]
    relation_id = relation_uri.rsplit("/", 1)[-1]
    
    assert str(relation_id).isdigit(), f"Invalid relation ID: {relation_id}"
    
    return {"name": target_name, "relation_id": relation_id}

## get_sushi_tag_count@v1
from geo_voyager.control_primitives import call_service
import json


def get_sushi_tag_count(intent_target=None, previous_observations=None, intent_text="", key="cuisine", value="sushi"):
    """Get the total usage count of cuisine=sushi in OSM."""
    params = {"key": key, "value": value}
    response = call_service("taginfo", path="/api/4/tag/stats", params=params)
    data = json.loads(response)
    
    if not data.get("data"):
        raise ValueError("No data found for cuisine=sushi")
        
    stats = data["data"][0]
    all_count = None
    
    # The "all" field in tag/stats response is a list of stats
    # Each stat has "type" and "count". We look for type "all".
    # Based on typical taginfo API, "all" is a list.
    # Let's handle both list and dict cases for robustness, but prioritize the list structure described in contract.
    
    all_stats = stats.get("all", [])
    if isinstance(all_stats, list):
        for stat in all_stats:
            if isinstance(stat, dict) and stat.get("type") == "all":
                all_count = stat.get("count")
                break
    elif isinstance(all_stats, dict):
        # If "all" is a dict directly, it might have "count" or "type"
        if all_stats.get("type") == "all":
            all_count = all_stats.get("count")
        else:
            # Maybe the whole "all" dict is the stat for "all" type?
            # Or maybe it's nested differently. Let's check if it has count directly.
            if "count" in all_stats:
                all_count = all_stats.get("count")
                
    if all_count is None:
        # Fallback: if "all" is not found, maybe "all" key is missing or structure is unexpected.
        # Try to find any stat with type "all" in the top level if "all" key is missing?
        # Or maybe the response structure is different.
        # Let's check if "all" key exists and is a list, but no "all" type found.
        # In some taginfo versions, "all" might be a single object if only one type exists?
        # But contract says: payload["data"] の要素のうち type が "all" のものの count が全体の使用数。
        # So we must find it. If not found, it's an error.
        # Check if "all" is missing entirely and try to find "count" at the top level of stats
        if "count" in stats:
            all_count = stats.get("count")
        else:
            raise ValueError("Count for type 'all' not found")
        
    return {"key": key, "value": value, "count": all_count}

def get_tokyo_ward_names_and_ids():
    """Get Japanese names and OSM relation IDs for all 23 wards of Tokyo."""
    from geo_voyager.control_primitives import call_service
    import json

    query = """
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    
    SELECT ?label ?relation WHERE {
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      ?ward gs:osmRelation ?relation .
    }
    ORDER BY ?label
    """
    response_text = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    data = json.loads(response_text)
    
    wards = []
    for row in data["results"]["bindings"]:
        label = row["label"]["value"]
        relation_uri = row["relation"]["value"]
        relation_id = relation_uri.rsplit("/", 1)[-1]
        assert str(relation_id).isdigit(), f"Invalid relation ID: {relation_id}"
        wards.append({"name": label, "relation_id": int(relation_id)})
        
    assert len(wards) == 23, f"Expected 23 wards, got {len(wards)}"
    return wards

## least_populous_area@v1
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def least_populous_area(dataset_id, area=None):
    """Find the administrative area with the smallest population, in all areas or in one named set (for example 東京都23区). Returns its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        row = load_admin_units(dataset_id, connection, area=area).order("population ASC, code5 ASC").limit(1).fetchone()
    if row is None:
        raise ValueError("No administrative area found")
    code5, name, population = row
    if type(population) is not int:
        raise TypeError("population must be an integer")
    return {"code5": code5, "name": name, "population": population}

from geo_voyager.control_primitives import connect_duckdb, load_stations


def northernmost_station(dataset_id):
    """Find the station with the largest latitude in the station dataset, with no filtering by operating state. Returns its name, latitude and longitude."""
    with connect_duckdb() as connection:
        row = load_stations(dataset_id, connection).order("latitude DESC, name ASC").limit(1).fetchone()
    if row is None:
        raise ValueError("No station found")
    name, latitude, longitude = row
    return {"name": name, "latitude": latitude, "longitude": longitude}

## top_areas_by_population@v1
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def top_areas_by_population(dataset_id, count, area=None):
    """List the given number of administrative areas with the largest populations, largest first, in all areas or in one named set (for example 東京都23区). Each item has its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        rows = load_admin_units(dataset_id, connection, area=area).order("population DESC, code5 ASC").limit(count).fetchall()
    if len(rows) != count:
        raise ValueError(f"{count} areas are required, {len(rows)} found")
    if any(type(population) is not int for _, _, population in rows):
        raise TypeError("population must be an integer")
    return [{"code5": code5, "name": name, "population": population} for code5, name, population in rows]

def top_cuisine_values(key="cuisine", count=3):
    """Get top 3 values for cuisine key by usage count."""

## total_population@v1
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
    with connect_duckdb() as connection:
        total = load_admin_units(dataset_id, connection, area=area).aggregate("sum(population) AS population_total").fetchone()[0]
    if type(total) is not int:
        raise TypeError("The population total must be an integer")
    return total
