## compare_cafe_and_restaurant_counts@v1
import json


def compare_cafe_and_restaurant_counts(intent_target, previous_observations=None, intent_text=None, cafe_tag="amenity=cafe", restaurant_tag="amenity=restaurant"):
    """実行時に指定された対象の前段 Observation から amenity=cafe と amenity=restaurant の件数を比較し、どちらが多いかを明示する JSON object を返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    candidates = []
    for value in decoded:
        candidates += value if isinstance(value, list) else [value]
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    assert matches
    target = matches[0]
    measurements = [o for o in candidates if isinstance(o, dict) and str(o.get(intent_target["id_type"])) == intent_target["id_value"] and o.get("tag") and "count" in o]
    cafe = [o for o in measurements if o.get("tag") == cafe_tag]
    restaurant = [o for o in measurements if o.get("tag") == restaurant_tag]
    assert cafe and restaurant
    cafe_count = int(cafe[0]["count"])
    restaurant_count = int(restaurant[0]["count"])
    larger = "cafe" if cafe_count > restaurant_count else "restaurant"
    return {
        "name": target["name"],
        "relation_id": target["relation_id"],
        "cafe_count": cafe_count,
        "restaurant_count": restaurant_count,
        "larger": larger,
    }

## compare_cafe_counts_by_name@v1
import json


def compare_cafe_counts_by_name(intent_target, previous_observations=None, intent_text=None, tag="amenity=cafe", first_name="渋谷区", second_name="新宿区"):
    """前段 Observation から実行時に解決された対象の amenity=cafe 件数を取得し、件数の大きい方を winner として shibuya_count、shinjuku_count、comparison を持つ JSON object を返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    measurements = [o for o in decoded if isinstance(o, dict) and o.get("tag") == tag and "count" in o]
    first = [o for o in measurements if o.get("name") == first_name]
    second = [o for o in measurements if o.get("name") == second_name]
    assert len(first) == 1 and len(second) == 1
    first_count = int(first[0]["count"])
    second_count = int(second[0]["count"])
    if first_count > second_count:
        winner = first_name
        comparison = f"{first_name} > {second_name}"
    elif second_count > first_count:
        winner = second_name
        comparison = f"{second_name} > {first_name}"
    else:
        winner = "tie"
        comparison = f"{first_name} == {second_name}"
    return {
        "winner": winner,
        "shibuya_count": first_count,
        "shinjuku_count": second_count,
        "comparison": comparison,
    }

    """taginfo の /api/4/tag/stats で2つのタグの使用数を取得し、どちらが多いかを明示する JSON object を返す。"""

            "taginfo",
            path="/api/4/tag/stats",
            params={"key": key, "value": value},
        ))

        assert data, "taginfo returned no data"
        all_rows = [row for row in data if row.get("type") == "all"]

        return int(all_rows[0]["count"])

    count2 = get_count(tag2_key, tag2_value)
    larger = f"{tag1_key}={tag1_value}" if count1 > count2 else f"{tag2_key}={tag2_value}"
    return {
        "tag1": f"{tag1_key}={tag1_value}",
        "tag1_count": count1,
        "tag2": f"{tag2_key}={tag2_value}",
        "tag2_count": count2,
        "larger": larger,

def count_ward_individuals(endpoint="/geo/sparql"):
    """YuisekinGeoSPARQL の pinned geographic graph に登録されている gs:Ward 型の個体の件数を整数で返す。"""
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "SELECT (COUNT(DISTINCT ?ward) AS ?count) WHERE { "
        "?ward a gs:Ward . "
        "}"
    )
    response = call_service("yuisekin-geosparql", path=endpoint, params={"query": query})
    payload = json.loads(response)
    rows = payload["results"]["bindings"]
    if not rows:
        raise ValueError("No count returned")
    count = int(rows[0]["count"]["value"])
    assert count >= 0
    return {"count": count}

## get_relation_id@v1 (seed)
import json

from geo_voyager.control_primitives import call_service


def get_relation_id(intent_target):
    """Find the OSM relation of a named place (for example a ward) with Nominatim and return its name and relation_id. When the target already carries a relation_id, that id is returned without a search."""
    if intent_target.get("id_type") == "relation_id" and intent_target.get("id_value"):
        return {"name": intent_target["name"], "relation_id": str(intent_target["id_value"])}
    results = json.loads(call_service("nominatim", path="/search",
                                      params={"q": intent_target["name"], "format": "jsonv2", "limit": "10"}))
    relations = [item for item in results if item.get("osm_type") == "relation"]
    if not relations:
        raise ValueError(f"No OSM relation found for {intent_target['name']}")
    return {"name": intent_target["name"], "relation_id": str(relations[0]["osm_id"])}

## get_station_stats@v1
from geo_voyager.control_primitives import connect_duckdb, load_stations


def get_station_stats(dataset_id, top_n=5):
    """駅データセットから重複排除後の件数、緯度上位N件、および緯度最大駅を取得する。"""
    with connect_duckdb() as connection:
        stations = load_stations(dataset_id, connection)
        
        # 重複排除した全件数
        total_count = stations.aggregate("COUNT(*)").fetchone()[0]
        
        # 緯度降順の上位N件
        top_rows = stations.order("latitude DESC").limit(top_n).fetchall()
        top5 = []
        for name, lat, lon in top_rows:
            # 一意な駅IDとして、名称と座標を結合したハッシュや文字列を使用
            # 元のスキーマにstation_idがなかったため、name_lat_lon をIDとする
            station_id = f"{name}_{lat}_{lon}"
            top5.append({
                "name": name,
                "station_id": station_id,
                "latitude": lat
            })
            
        # 緯度が最大の駅
        max_row = stations.order("latitude DESC, name ASC").limit(1).fetchone()
        if max_row is None:
            raise ValueError("No station found")
        
        name, lat, lon = max_row
        station_id = f"{name}_{lat}_{lon}"
        
        max_station = {
            "name": name,
            "station_id": station_id,
            "latitude": lat
        }
        
    return {
        "total_count": total_count,
        "top5": top5,
        "max": max_station
    }

## get_ward_relation_id@v1
import json
from geo_voyager.control_primitives import call_service


def get_ward_relation_id(intent_target, endpoint="/geo/sparql"):
    """実行時に指定された対象の gs:Ward 個体の日本語 rdfs:label と gs:osmRelation を取得し、末尾の数字のみを relation_id として返す。"""
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> "
        "SELECT ?label ?relation WHERE { "
        "?ward a gs:Ward ; rdfs:label ?label ; gs:osmRelation ?relation . "
        f'FILTER(LANG(?label) = "ja" && STR(?label) = "{intent_target["name"]}") '
        "} ORDER BY ?label"
    )
    response = call_service("yuisekin-geosparql", path=endpoint, params={"query": query})
    payload = json.loads(response)
    rows = payload["results"]["bindings"]
    if not rows:
        raise ValueError(f"No gs:Ward found for {intent_target['name']}")
    label = rows[0]["label"]["value"]
    relation = rows[0]["relation"]["value"]
    relation_id = relation.rsplit("/", 1)[-1]
    assert str(relation_id).isdigit()
    return {"name": label, "relation_id": relation_id}

## get_wards_population_with_relation@v1
import json
from geo_voyager.control_primitives import call_service


def get_wards_population_with_relation(dataset_id, previous_observations, intent_text, endpoint="/api/interpreter"):
    """
    東京23区の各行政区の名称とOSM関係IDを用いて、Overpass APIから人口を取得し、
    name, relation_id, population を持つJSONオブジェクトの一覧を返す。
    """
    wards = json.loads(previous_observations[0])
    
    results = []
    for ward in wards:
        name = ward["name"]
        relation_id = ward["relation_id"]
        
        # OSM relation ID から Overpass area ID に変換 (int(osm_id) + 3600000000)
        area_id = int(relation_id) + 3600000000
        
        # Overpass QL を構築
        # [out:json];
        # area:<area_id>;
        # nwr["name"="name"];
        # out body;
        # 
        # 注意: 問題文の制約「Verify the resolved area exists before counting」に従い、
        # まず area の存在確認と、その後の population の取得を行う。
        # ただし、Overpass の構文として、area を定義してから要素を取得する必要がある。
        # nwr["name"="name"](area:<area_id>); out body; という形にする。
        
        query = f'[out:json];nwr["name"="{name}"](area:{area_id});out body;'
        
        response = call_service("overpass", path=endpoint, params={"data": query})
        data = json.loads(response)
        
        # エラーチェック
        if "remark" in data:
            raise ValueError(f"Overpass error for {name}: {data['remark']}")
        
        elements = data.get("elements", [])
        
        # 人口を取得
        population = None
        for el in elements:
            tags = el.get("tags", {})
            if "population" in tags:
                population = tags["population"]
                break
                
        if population is None:
             raise ValueError(f"Population not found for ward: {name}")
             
        results.append({
            "name": name,
            "relation_id": relation_id,
            "population": int(population)
        })
        
    return results

## sum_previous_ward_populations@v1
import json


def sum_previous_ward_populations(previous_observations, intent_text):
    """前段のObservationから対象と人口測定をIDで対応付け、全対象の人口を整数で合計してtotal_populationを返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    targets = [item for item in decoded if isinstance(item, list) and item and all(isinstance(x, dict) and "name" in x and "relation_id" in x and "population" not in x for x in item)]
    measurements = [item for item in decoded if isinstance(item, list) and item and all(isinstance(x, dict) and "name" in x and "relation_id" in x and "population" in x for x in item)]
    if not targets or not measurements:
        raise ValueError("target list and population measurements are required")
    target_ids = [str(t["relation_id"]) for t in targets[0]]
    if len(set(target_ids)) != len(target_ids):
        raise ValueError("target relation_id must be unique")
    measurement_ids = [str(m["relation_id"]) for m in measurements[0]]
    if len(set(measurement_ids)) != len(measurement_ids):
        raise ValueError("measurement relation_id must be unique")
    if set(target_ids) != set(measurement_ids):
        raise ValueError("all targets must have population measurements")
    total_population = 0
    for target in targets[0]:
        measurement = next(m for m in measurements[0] if str(m["relation_id"]) == str(target["relation_id"]))
        population = measurement["population"]
        if type(population) is not int:
            raise TypeError("population must be an integer")
        total_population += population
    return {"total_population": total_population}

## total_population@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
    with connect_duckdb() as connection:
        total = load_admin_units(dataset_id, connection, area=area).aggregate("sum(population) AS population_total").fetchone()[0]
