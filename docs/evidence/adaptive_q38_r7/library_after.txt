## get_all_wards_with_relation_ids@v1
import json
from geo_voyager.control_primitives import call_service


def get_all_wards_with_relation_ids():
    """YuisekinGeoSPARQL の pinned geographic graph で gs:Ward 型の全個体を取得し、日本語 rdfs:label を name、gs:osmRelation の末尾数値を relation_id として JSON 配列で返す。同時に個体の件数 count を整数で返す。"""
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> "
        "SELECT ?label ?relation WHERE { "
        "?ward a gs:Ward ; rdfs:label ?label ; gs:osmRelation ?relation . "
        'FILTER(LANG(?label) = "ja") '
        "} ORDER BY ?label"
    )
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    payload = json.loads(response)
    rows = payload["results"]["bindings"]
    if not rows:
        raise ValueError("No gs:Ward entities found")
    result = []
    for row in rows:
        label = row["label"]["value"]
        relation = row["relation"]["value"]
        relation_id = relation.rsplit("/", 1)[-1]
        assert str(relation_id).isdigit()
        result.append({"name": label, "relation_id": relation_id})
    return {"wards": result, "count": len(result)}

    """出発点と到着点の徒歩経路の距離・時間・平均速度・区間合計・整合性判定を返す。"""
    body = json.dumps({"locations": [{"lat": origin_lat, "lon": origin_lon}, {"lat": destination_lat, "lon": destination_lon}], "costing": costing, "units": "kilometers"})

    if not trip:
        raise ValueError("No trip in Valhalla response")
    summary = trip.get("summary")
    if not summary:

    legs = trip.get("legs")
    if not legs:
        raise ValueError("No legs in Valhalla response")
    leg_distance_km = sum(float(leg["summary"]["length"]) for leg in legs)
    leg_time_s = sum(float(leg["summary"]["time"]) for leg in legs)

    return {
        "distance_km": distance_km,

        "leg_distance_km": leg_distance_km,
        "leg_time_s": leg_time_s,
        "avg_speed_formula": "distance_km / (time_s / 3600)",
        "normal_speed_range_kmh": [4.0, 5.0],

        unique_rows = set()

            latitude = row[1]
            longitude = row[2]
            unique_rows.add((name, latitude, longitude))

        top5 = []
        for row in sorted_list[:5]:
            name, latitude, longitude = row
            station_id = hashlib.sha256(f"{name}|{latitude}|{longitude}".encode("utf-8")).hexdigest()

                "name": name,
                "station_id": station_id,
                "latitude": latitude
            })

        max_station_id = hashlib.sha256(f"{max_name}|{max_latitude}|{max_longitude}".encode("utf-8")).hexdigest()
        max_info = {
            "name": max_name,
            "station_id": max_station_id,
            "latitude": max_latitude
        }

        return {
            "total_count": total_count,
            "top5": top5,
            "max": max_info
        }

def sum_ward_population(previous_observations, intent_text):
    """前段 Observation の区 population を合計し、total_population を整数で返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    records = [
        item for obj in decoded
        if isinstance(obj, list)
        for item in obj
        if isinstance(item, dict) and "name" in item and "relation_id" in item and "population" in item
    ]
    assert len(records) == 23, "Expected 23 ward population records"
    assert len({str(item["relation_id"]) for item in records}) == 23, "relation_id must be unique"

    assert isinstance(total, int), "total_population must be an integer"
    return {"total_population": total}

def tag_usage_count(key, value):
    """Count how many times the OSM tag key=value is used in the whole OSM database, with Taginfo's tag statistics. Returns the tag and the count."""
    payload = json.loads(call_service("taginfo", path="/api/4/tag/stats", params={"key": key, "value": value}))
    total = [item["count"] for item in payload["data"] if item["type"] == "all"]
    if len(total) != 1:
        raise ValueError(f"Taginfo returned no total for {key}={value}")
    return {"key": key, "value": value, "count": total[0]}
