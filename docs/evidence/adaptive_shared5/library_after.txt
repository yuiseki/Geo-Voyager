from geo_voyager.control_primitives import call_service
import json


def check_osm_tags_for_wards(intent_target, previous_observations=None):
    """各行政区画について、Overpass API を使用して OSM relation のタグ情報（人口関連タグの有無）を確認し、名称と ID およびタグ情報を返す。"""
    if previous_observations is None:
        raise ValueError("previous_observations is required")
    
    wards = json.loads(previous_observations[0])
    if not isinstance(wards, list) or len(wards) != 23:
        raise ValueError("Expected 23 wards in previous_observations")
    
    results = []
    for ward in wards:
        name = ward["name"]
        relation_id = ward["relation_id"]
        area_id = int(relation_id) + 3600000000
        
        query = f"[out:json][timeout:12];nwr({area_id});out tags;"
        params = {"data": query}
        response_text = call_service("overpass", path="/api/interpreter", params=params)
        payload = json.loads(response_text)
        
        elements = payload.get("elements", [])
        tags = {}
        for element in elements:
            if element.get("type") == "relation":
                tags = element.get("tags", {})
                break
        
        has_population_tag = any("population" in key for key in tags.keys())
        results.append({
            "name": name,
            "relation_id": relation_id,
            "tags": tags,
            "has_population_tag": has_population_tag
        })
    
    return results

from geo_voyager.control_primitives import call_service
import json


def get_all_ward_ids_and_names():
    """Fetch all Tokyo 23 wards' Japanese labels and OSM relation IDs."""
    query = """
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    
    SELECT ?label ?relation
    WHERE {
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      ?ward gs:osmRelation ?relation .
    }
    ORDER BY ?label
    """
    payload = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    data = json.loads(payload)
    bindings = data.get("results", {}).get("bindings", [])
    
    if not bindings:
        raise ValueError("No wards found")
    
    results = []
    for row in bindings:
        label = row.get("label", {}).get("value", "")
        relation_uri = row.get("relation", {}).get("value", "")
        
        if not label or not relation_uri:
            continue
            
        parts = relation_uri.rsplit("/", 1)
        if len(parts) > 1:
            candidate_id = parts[-1]
            if str(candidate_id).isdigit():
                results.append({"name": label, "relation_id": int(candidate_id)})
    
    assert len(results) == 23, f"Expected 23 wards, found {len(results)}"
    return results

## get_relation_id@v1
import json

from geo_voyager.control_primitives import call_service


def get_relation_id(intent_target):
    """Find the OSM relation of a named place (for example a ward) with Nominatim and return its name and relation_id. When the target already carries a relation_id, that id is returned without a search."""
    if intent_target.get("id_type") == "relation_id" and intent_target.get("id_value"):
        return {"name": intent_target["name"], "relation_id": str(intent_target["id_value"])}
    results = json.loads(call_service("nominatim", path="/search",
                                      params={"q": intent_target["name"], "format": "jsonv2", "limit": "10"}))
    relations = [item for item in results if item.get("osm_type") == "relation"]
    if not relations:
        raise ValueError(f"No OSM relation found for {intent_target['name']}")
    return {"name": intent_target["name"], "relation_id": str(relations[0]["osm_id"])}

def get_ward_relation_id():

    service_id = "yuisekin-geosparql"
    sparql_query = """
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    
    SELECT ?label ?relation
    WHERE {
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      ?ward gs:osmRelation ?relation .
    }
    ORDER BY ?label
    """
    
    params = {
        "query": sparql_query
    }
    
    response_text = call_service(service_id, path="/geo/sparql", params=params)
    import json
    payload = json.loads(response_text)
    
    bindings = payload.get("results", {}).get("bindings", [])
    
    target_name = "台東区"
    relation_id = None
    
    for row in bindings:
        label_val = row.get("label", {}).get("value", "")
        relation_val = row.get("relation", {}).get("value", "")
        
        if label_val == target_name:
            # relation_val is like "http://www.openstreetmap.org/relation/12345"
            # Extract the ID part
            if relation_val:
                parts = relation_val.rsplit("/", 1)
                if len(parts) > 1:
                    candidate_id = parts[-1]
                    if str(candidate_id).isdigit():
                        relation_id = candidate_id
                        break
    
    assert relation_id is not None, f"Relation ID for {target_name} not found"
    assert str(relation_id).isdigit(), f"Relation ID is not a digit: {relation_id}"
    
    return relation_id

## route_summary@v1
import json

from geo_voyager.control_primitives import call_service


def route_summary(origin_lat, origin_lon, destination_lat, destination_lon, costing="auto"):
    """Route between two points with Valhalla and return the length in kilometres and the time in seconds and minutes. costing is auto, pedestrian or bicycle."""
    request = {"locations": [{"lat": origin_lat, "lon": origin_lon}, {"lat": destination_lat, "lon": destination_lon}],
               "costing": costing, "units": "kilometers"}
    summary = json.loads(call_service("valhalla", path="/route", body=json.dumps(request),
                                      content_type="application/json"))["trip"]["summary"]
    return {"costing": costing, "length_km": summary["length"], "time_s": summary["time"], "time_min": summary["time"] / 60}

def tag_usage_count(key, value):
    """Count how many times the OSM tag key=value is used in the whole OSM database, with Taginfo's tag statistics. Returns the tag and the count."""
    payload = json.loads(call_service("taginfo", path="/api/4/tag/stats", params={"key": key, "value": value}))
    total = [item["count"] for item in payload["data"] if item["type"] == "all"]
    if len(total) != 1:
        raise ValueError(f"Taginfo returned no total for {key}={value}")
    return {"key": key, "value": value, "count": total[0]}

## walking_time_minutes@v1
from geo_voyager.control_primitives import call_service
import json


def walking_time_minutes(previous_observations=None, intent_text="2904.379秒を分単位に換算した徒歩所要時間"):
    """Convert walking time from seconds to minutes."""
    if not previous_observations:
        raise ValueError("previous_observations is required")
    
    # Parse the previous observation to get the walking time in seconds
    try:
        obs = json.loads(previous_observations[0])
        time_s = obs["time_s"]
    except (IndexError, json.JSONDecodeError, KeyError) as e:
        raise ValueError(f"Failed to parse previous observation: {e}")
    
    # Convert seconds to minutes
    time_min = time_s / 60
    
    return {"time_min": time_min}
