from geo_voyager.control_primitives import call_service
import json


def calculate_route_distance(intent_target, previous_observations=None):
    """Calculate the driving distance between two coordinates using valhalla."""
    start_lat = 35.6580
    start_lon = 139.7016
    end_lat = 35.6896
    end_lon = 139.7006

    body = json.dumps({
        "locations": [
            {"lat": start_lat, "lon": start_lon},
            {"lat": end_lat, "lon": end_lon}
        ],
        "costing": "auto",
        "units": "kilometers"
    })

    response_text = call_service(service_id="valhalla", path="/route", body=body, content_type="application/json")
    response = json.loads(response_text)

    trip = response.get("trip")
    if not trip:
        raise ValueError("No trip found in response")

    length_km = trip["summary"]["length"]

    return {"distance_km": length_km}

## compare_amenity_counts@v1
from geo_voyager.control_primitives import call_service


def compare_amenity_counts(intent_target, previous_observations=None):
    """Compare counts of restaurants and cafes in the runtime target area."""
    if previous_observations is None:
        previous_observations = []
    
    from geo_voyager.control_primitives import call_service
    import json

    def get_count(tag):
        candidates = []
        for text in previous_observations:
            value = json.loads(text)
            candidates += value if isinstance(value, list) else [value]
        matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
        if not matches:
            raise ValueError("Target not found in previous observations")
        target = matches[0]
        osm_id = int(target["relation_id"])
        area_id = osm_id + 3600000000
        ql = f"[out:json][timeout:12];nwr[\"{tag.split('=')[0]}\"=\"{tag.split('=')[1]}\"](area:{area_id});out count;"
        response = call_service("overpass", path="/api/interpreter", body=ql, content_type="text/plain")
        data = json.loads(response)
        if "elements" not in data or len(data["elements"]) == 0:
            raise ValueError("No count element returned")
        return int(data["elements"][0]["tags"]["total"])

    restaurant_count = get_count("amenity=restaurant")
    cafe_count = get_count("amenity=cafe")
    
    if restaurant_count > cafe_count:
        return {"name": intent_target["name"], "relation_id": intent_target["id_value"], "restaurant_count": restaurant_count, "cafe_count": cafe_count, "comparison": "restaurant_more"}
    else:
        return {"name": intent_target["name"], "relation_id": intent_target["id_value"], "restaurant_count": restaurant_count, "cafe_count": cafe_count, "comparison": "cafe_more_or_equal"}

def compare_ramen_sushi_counts(intent_target, previous_observations=None):
    """Compare usage counts of ramen and sushi from previous observations."""
    if previous_observations is None:
        raise ValueError("previous_observations is required")
    
    decoded = [json.loads(text) for text in previous_observations]
    
    ramen_data = None
    sushi_data = None
    
    for obs in decoded:
        if isinstance(obs, dict):
            if obs.get("value") == "ramen" and "count" in obs:
                ramen_data = obs
            elif obs.get("value") == "sushi" and "count" in obs:
                sushi_data = obs
                
    if ramen_data is None:
        raise ValueError("Ramen count data not found in previous observations")
    if sushi_data is None:
        raise ValueError("Sushi count data not found in previous observations")
        
    ramen_count = ramen_data["count"]
    sushi_count = sushi_data["count"]
    
    if ramen_count > sushi_count:
        return {"winner": "ramen", "ramen_count": ramen_count, "sushi_count": sushi_count}
    else:
        return {"winner": "sushi", "ramen_count": ramen_count, "sushi_count": sushi_count}

from geo_voyager.control_primitives import call_service
import json


def get_boundingbox(intent_target, **kwargs):
    """実行時に指定された対象について、Nominatim検索で境界ボックスを取得する。"""
    name = intent_target.get("name")
    if not name:
        raise ValueError("Target name is required")
    
    response = call_service(
        service_id="nominatim",
        path="/search",
        params={"q": name, "format": "jsonv2", "limit": "1"}
    )
    results = json.loads(response)
    
    if not results:
        raise ValueError(f"No results found for {name}")
    
    item = results[0]
    boundingbox = item.get("boundingbox")
    if not boundingbox:
        raise ValueError(f"No boundingbox found for {name}")
    
    osm_id = item.get("osm_id")
    osm_type = item.get("osm_type")
    relation_id = f"{osm_type}/{osm_id}" if osm_type and osm_id else None
    
    return {
        "name": name,
        "relation_id": relation_id,
        "boundingbox": boundingbox
    }

## get_osm_relation_id@v1
from geo_voyager.control_primitives import call_service
import json


def get_osm_relation_id(intent_target, **kwargs):
    """Get the OSM relation ID for the target specified at runtime."""
    target_name = intent_target["name"]
    query = f"""
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

    SELECT ?label ?relation
    WHERE {{
      ?ward a gs:Ward .
      ?ward rdfs:label ?label .
      FILTER(LANG(?label) = "ja")
      FILTER(STR(?label) = "{target_name}")
      ?ward gs:osmRelation ?relation .
    }}
    ORDER BY ?label
    """
    response = call_service(service_id="yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    results = json.loads(response)
    bindings = results["results"]["bindings"]
    
    if not bindings:
        raise ValueError(f"No ward found with name {target_name}")
    
    binding = bindings[0]
    relation = binding["relation"]["value"]
    relation_id = relation.rsplit("/", 1)[-1]
    
    assert str(relation_id).isdigit(), f"Relation ID is not a digit: {relation_id}"
    
    return {"name": target_name, "relation_id": relation_id}

def get_southernmost_latitude(intent_target, previous_observations):
    """Get the southernmost latitude of the bounding box for the runtime target."""
    candidates = []
    for text in previous_observations:
        value = json.loads(text)
        candidates += value if isinstance(value, list) else [value]
    
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    if not matches:
        raise ValueError("Target not found in previous observations")
    
    target = matches[0]
    boundingbox = target.get("boundingbox")
    if not boundingbox:
        raise ValueError("Bounding box not found")
    
    southern_lat = boundingbox[0]
    
    return {
        "name": target["name"],
        "relation_id": target["relation_id"],
        "southernmost_latitude": float(southern_lat)
    }

    # Using /api/4/tag/stats to get the exact count for a specific key=value pair
    service_id = "taginfo"
    path = "/api/4/tag/stats"
    params = {
        "key": "cuisine",
        "value": "sushi"
    }
    
    response_text = call_service(service_id, path=path, params=params)
    payload = json.loads(response_text)
    
    # The response structure for /api/4/tag/stats is {"data": [{"type": "all", "count": ...}, ...]}
    # We need to find the entry with type "all" and get its count
    data = payload.get("data", [])
    
    count = 0
    for item in data:
        if item.get("type") == "all":
            count = item.get("count", 0)
            break
            
    return {
        "key": "cuisine",
        "value": "sushi",
        "count": count
    }
