from geo_voyager.control_primitives import call_service
import json


def calculate_route_distance_km(intent_target, start_lat=35.6580, start_lon=139.7016, end_lat=35.6896, end_lon=139.7006):
    """Calculate driving distance in kilometers between two coordinates using valhalla."""
    body = json.dumps({
        "locations": [
            {"lat": start_lat, "lon": start_lon},
            {"lat": end_lat, "lon": end_lon}
        ],
        "costing": "auto",
        "units": "kilometers"
    })
    response = call_service("valhalla", path="/route", body=body, content_type="application/json")
    data = json.loads(response)
    trip = data.get("trip", {})
    summary = trip.get("summary", {})
    distance_km = summary.get("length", 0.0)
    return {"distance_km": distance_km}

## compare_amenity_counts@v1
from geo_voyager.control_primitives import call_service
import json


def compare_amenity_counts(intent_target, previous_observations=None, intent_text=""):
    """Compare counts of cafe and restaurant in the runtime target area."""
    if previous_observations is None:
        previous_observations = []
    
    candidates = []
    for text in previous_observations:
        value = json.loads(text)
        candidates += value if isinstance(value, list) else [value]
    
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    if not matches:
        raise ValueError("Target not found in previous observations")
    
    target = matches[0]
    
    cafe_obs = [o for o in candidates if isinstance(o, dict) and o.get("tag") == "amenity=cafe" and "count" in o]
    restaurant_obs = [o for o in candidates if isinstance(o, dict) and o.get("tag") == "amenity=restaurant" and "count" in o]
    
    if not cafe_obs or not restaurant_obs:
        raise ValueError("Missing cafe or restaurant observation")
    
    cafe_count = cafe_obs[0]["count"]
    restaurant_count = restaurant_obs[0]["count"]
    
    if cafe_count > restaurant_count:
        winner = "cafe"
        winner_count = cafe_count
    else:
        winner = "restaurant"
        winner_count = restaurant_count
        
    return {
        "name": target["name"],
        "relation_id": target["relation_id"],
        "cafe_count": cafe_count,
        "restaurant_count": restaurant_count,
        "winner": winner,
        "winner_count": winner_count
    }

import json


def count_4char_wards(previous_observations, intent_text):
    """Extract wards with 4-character Japanese names from previous observations and return the count."""
    if not previous_observations:
        raise ValueError("No previous observations provided")
    
    try:
        wards = json.loads(previous_observations[0])
    except (json.JSONDecodeError, IndexError):
        raise ValueError("Failed to parse previous observations")
    
    filtered_wards = [w for w in wards if isinstance(w, dict) and len(w.get("name", "")) == 4]
    
    if not filtered_wards:
        raise ValueError("No wards with 4-character names found")
        
    return {"count": len(filtered_wards)}

from geo_voyager.control_primitives import call_service


def get_all_ward_names_and_relation_ids(intent_target, name=None):
    """Fetch all wards' Japanese names and OSM relation IDs."""
    service_id = "yuisekin-geosparql"
    
    query = """
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    
    SELECT ?label ?relation
    WHERE {
        ?ward a gs:Ward .
        ?ward rdfs:label ?label .
        FILTER(LANG(?label) = "ja")
        ?ward gs:osmRelation ?relation .
    }
    ORDER BY ?label
    """
    
    response_text = call_service(
        service_id,
        path="/geo/sparql",
        params={"query": query}
    )

    import json
    response_json = json.loads(response_text)

    results = []
    bindings = response_json.get("results", {}).get("bindings", [])
    
    for binding in bindings:
        label_value = binding.get("label", {}).get("value", "")
        relation_value = binding.get("relation", {}).get("value", "")
        
        if relation_value:
            relation_id_str = relation_value.rsplit("/", 1)[-1]
            assert relation_id_str.isdigit(), f"Relation ID is not a digit: {relation_id_str}"
            results.append({
                "name": label_value,
                "relation_id": relation_id_str
            })
            
    return results

    osm_id = item.get("osm_id")
    osm_type = item.get("osm_type")
    if osm_type == "relation" and osm_id:
        return {"name": name, "relation_id": str(osm_id)}
    elif osm_type == "node" or osm_type == "way":
        # Try to find parent relation if not directly a relation
        # Nominatim search doesn't always give parent relation directly.
        # We might need to fallback or raise if not found.
        # For this intent, we assume the search result contains the relation or we can derive it.
        # If it's a node/way, we might not get the relation ID directly from search.
        # However, the prompt asks to get OSM relation ID.
        # Let's check if 'extratags' gave us a relation.
        extratags = item.get("extratags", {})
        if "osm_relation" in extratags:

        # But for simplicity and given the previous skills used SPARQL, let's stick to Nominatim as requested.
        # If Nominatim doesn't provide it, we might need to raise.
        raise ValueError("Could not find OSM relation ID for target")
    else:
        raise ValueError("Could not find OSM relation ID for target")

## get_ward_names_and_ids@v1
from geo_voyager.control_primitives import call_service
import json


def get_ward_names_and_ids(intent_target, name=None):
    """Fetch all wards' Japanese names and OSM relation IDs."""
    service_id = "yuisekin-geosparql"
    
    # Construct SPARQL query
    # We need to select ?label (Japanese) and ?relation (gs:osmRelation)
    # Filter by type gs:Ward and language of label is ja
    # Order by label
    
    query = """
    PREFIX gs: <https://yuiseki.net/geosparql/schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    
    SELECT ?label ?relation
    WHERE {
        ?ward a gs:Ward .
        ?ward rdfs:label ?label .
        FILTER(LANG(?label) = "ja")
        ?ward gs:osmRelation ?relation .
    }
    ORDER BY ?label
    """
    
    # Call the service
    # path="/geo/sparql", params={"query": query}
    try:
        response_text = call_service(
            service_id,
            path="/geo/sparql",
            params={"query": query}
        )
    except Exception as e:
        raise RuntimeError(f"Failed to call SPARQL service: {e}")

    try:
        response_json = json.loads(response_text)
    except json.JSONDecodeError:
        raise RuntimeError(f"Failed to parse JSON response: {response_text}")

    results = []
    
    # Parse the response
    # The root is a dict, rows are in payload["results"]["bindings"]
    bindings = response_json.get("results", {}).get("bindings", [])
    
    for binding in bindings:
        # Extract label
        label_obj = binding.get("label", {})
        label_value = label_obj.get("value", "")
        
        # Extract relation
        relation_obj = binding.get("relation", {})
        relation_value = relation_obj.get("value", "")
        
        # Extract the numeric ID from the relation URI
        # The relation URI ends with the numeric ID
        # Example: http://www.openstreetmap.org/relation/12345
        if relation_value:
            try:
                # rsplit("/", 1) splits from the right, taking the last part
                relation_id_str = relation_value.rsplit("/", 1)[-1]
                # Assert it is a digit
                assert relation_id_str.isdigit(), f"Relation ID is not a digit: {relation_id_str}"
                relation_id = relation_id_str
            except (IndexError, AssertionError) as e:
                raise RuntimeError(f"Failed to extract relation ID: {e}")
        else:
            raise RuntimeError("Relation value is empty")
            
        results.append({
            "name": label_value,
            "relation_id": relation_id
        })
        
    return results

## walking_time_minutes@v1
import json


def walking_time_minutes(intent_target, previous_observations=None):
    """Convert walking time from seconds to minutes."""
    if not previous_observations:
        raise ValueError("No previous observations provided")
    
    decoded = [json.loads(text) for text in previous_observations]
    if not decoded:
        raise ValueError("No previous observations found")
    
    # Find the observation with time_seconds
    time_obs = None
    for obs in decoded:
        if "time_seconds" in obs:
            time_obs = obs
            break
    
    if time_obs is None:
        raise ValueError("No time_seconds found in previous observations")
    
    time_seconds = time_obs["time_seconds"]
    time_minutes = time_seconds / 60.0
    
    return {"time_minutes": time_minutes}
