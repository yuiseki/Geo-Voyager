from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def aggregate_tokyo23_population(dataset_id, area="東京都23区"):
    """東京23区の区別人口と全体合計を集計し、JSON object の一覧として返す。"""
    with connect_duckdb() as connection:
        rows = load_admin_units(dataset_id, connection, area=area).fetchall()
    
    results = []
    total_pop = 0
    
    for row in rows:
        # load_admin_units は (code5, name, population) を返す
        name = row[1]
        population = row[2]
        
        if population is None:
            raise ValueError(f"Population is missing for {name}")
            
        results.append({
            "name": name,
            "population": population
        })
        total_pop += population
        
    results.append({
        "name": "total",
        "population": total_pop
    })
    
    return results

## compare_cafe_restaurant_counts@v1
import json


def compare_cafe_restaurant_counts(intent_target, previous_observations, intent_text, tag_a="amenity=cafe", tag_b="amenity=restaurant"):
    """実行時に指定された対象の前段 Observation から cafe と restaurant の件数を比較し、winner を返す。"""
    decoded = [json.loads(text) for text in previous_observations]
    candidates = []
    for value in decoded:
        candidates += value if isinstance(value, list) else [value]
    matches = [t for t in candidates if isinstance(t, dict) and str(t.get(intent_target["id_type"])) == intent_target["id_value"]]
    assert matches
    target = matches[0]
    records = [o for o in candidates if isinstance(o, dict) and str(o.get(intent_target["id_type"])) == intent_target["id_value"] and o.get("tag") in (tag_a, tag_b) and "count" in o]
    assert len(records) == 2
    counts = {o["tag"]: o["count"] for o in records}
    assert tag_a in counts and tag_b in counts
    winner = tag_a if counts[tag_a] > counts[tag_b] else tag_b
    return {
        "name": target["name"],
        "relation_id": target["relation_id"],
        "tag_a": tag_a,
        "count_a": counts[tag_a],
        "tag_b": tag_b,
        "count_b": counts[tag_b],
        "winner": winner
    }

## get_all_ward_name_and_relation_id@v1
import json
from geo_voyager.control_primitives import call_service


def get_all_ward_name_and_relation_id():
    """YuisekinGeoSPARQL の全 gs:Ward について、日本語 rdfs:label を name として、必須 gs:osmRelation の末尾数値を relation_id として取得し、name と relation_id を持つ JSON object の一覧を返す。"""
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> "
        "SELECT ?label ?relation WHERE { "
        "?ward a gs:Ward ; rdfs:label ?label ; gs:osmRelation ?relation . "
        'FILTER(LANG(?label) = "ja") '
        "} ORDER BY ?label"
    )
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    payload = json.loads(response)
    bindings = payload["results"]["bindings"]
    assert len(bindings) == 23
    result = []
    for row in bindings:
        label = row["label"]["value"]
        relation = row["relation"]["value"]
        relation_id = relation.rsplit("/", 1)[-1]
        assert str(relation_id).isdigit()
        result.append({"name": label, "relation_id": relation_id})
    return result

from geo_voyager.control_primitives import call_service


def get_pedestrian_route_details(previous_observations=None, intent_text=None, origin_lat=35.6580, origin_lon=139.7016, destination_lat=35.6896, destination_lon=139.7006, costing="pedestrian"):
    """実行時に指定された出発点と到着点の座標から valhalla の pedestrian 経路を取得し、距離・所要時間・平均速度・座標列を含む非空の経路詳細を返す。"""
    time_result = get_pedestrian_route_time_minutes(origin_lat=origin_lat, origin_lon=origin_lon, destination_lat=destination_lat, destination_lon=destination_lon, costing=costing)
    body = json.dumps({"locations": [{"lat": origin_lat, "lon": origin_lon}, {"lat": destination_lat, "lon": destination_lon}], "costing": costing, "units": "kilometers"})
    payload = json.loads(call_service("valhalla", path="/route", body=body, content_type="application/json"))
    trip = payload.get("trip")
    if not trip:
        raise ValueError("Missing trip in valhalla response")
    summary = trip.get("summary")
    if not summary or "length" not in summary or "time" not in summary:
        raise ValueError("Missing summary length or time in valhalla trip")
    legs = trip.get("legs")
    if not legs:
        raise ValueError("Missing legs in valhalla trip")
    coordinates = []
    for leg in legs:
        shape = leg.get("shape")
        if not shape:
            raise ValueError("Missing shape in valhalla leg")
        coordinates.extend(shape)
    if not coordinates:
        raise ValueError("Missing coordinates in valhalla trip")
    distance_km = float(summary["length"])
    time_minutes = float(summary["time"]) / 60.0
    average_speed_kmh = distance_km / (time_minutes / 60.0) if time_minutes > 0 else 0.0
    previous_time = json.loads(previous_observations[0])["time_minutes"] if previous_observations else None
    valid = previous_time is not None and abs(float(previous_time) - time_minutes) <= 0.01
    return {"distance_km": distance_km, "time_minutes": time_minutes, "average_speed_kmh": average_speed_kmh, "coordinates": coordinates, "valid": valid}

## get_relation_id@v1 (seed)
import json

from geo_voyager.control_primitives import call_service


def get_relation_id(intent_target):
    """Find the OSM relation of a named place (for example a ward) with Nominatim and return its name and relation_id. When the target already carries a relation_id, that id is returned without a search."""
    if intent_target.get("id_type") == "relation_id" and intent_target.get("id_value"):
        return {"name": intent_target["name"], "relation_id": str(intent_target["id_value"])}
    results = json.loads(call_service("nominatim", path="/search",
                                      params={"q": intent_target["name"], "format": "jsonv2", "limit": "10"}))
    relations = [item for item in results if item.get("osm_type") == "relation"]
    if not relations:
        raise ValueError(f"No OSM relation found for {intent_target['name']}")
    return {"name": intent_target["name"], "relation_id": str(relations[0]["osm_id"])}

def get_ward_relation_id(name="渋谷区"):
    """yuisekin-geosparql の SPARQL で指定された gs:Ward 個体の日本語 rdfs:label と gs:osmRelation の末尾数値 relation_id を取得し、name と relation_id を持つ JSON object を1件返す。"""
    query = (
        "PREFIX gs: <https://yuiseki.net/geosparql/schema#> "
        "PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> "
        "SELECT ?label ?relation WHERE { "
        "?ward a gs:Ward ; rdfs:label ?label ; gs:osmRelation ?relation . "
        f'FILTER(LANG(?label) = "ja" && STR(?label) = "{name}") '
        "} ORDER BY ?label"
    )
    response = call_service("yuisekin-geosparql", path="/geo/sparql", params={"query": query})
    payload = json.loads(response)
    bindings = payload["results"]["bindings"]
    if len(bindings) != 1:
        raise ValueError(f"Expected exactly one gs:Ward for {name}, got {len(bindings)}")
    row = bindings[0]
    label = row["label"]["value"]
    relation = row["relation"]["value"]
    relation_id = relation.rsplit("/", 1)[-1]
    assert str(relation_id).isdigit()
    return {"name": label, "relation_id": relation_id}

## join_ward_population@v1
import json
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def join_ward_population(previous_observations, dataset_id):
    """前段の東京23区の name と relation_id の一覧と、Dataset の区単位の人口データを結合し、name、relation_id、population を持つ JSON object の配列を返す。"""
    wards_info = json.loads(previous_observations[0])
    ward_map = {w["name"]: w["relation_id"] for w in wards_info}
    
    with connect_duckdb() as connection:
        rows = load_admin_units(dataset_id, connection, area="東京都23区").fetchall()
        
    results = []
    for code5, name, population in rows:
        if name not in ward_map:
            raise ValueError(f"relation_id not found for ward: {name}")
        results.append({
            "name": name,
            "relation_id": ward_map[name],
            "population": population
        })
        
    return results

## least_populous_area@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def least_populous_area(dataset_id, area=None):
    """Find the administrative area with the smallest population, in all areas or in one named set (for example 東京都23区). Returns its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        row = load_admin_units(dataset_id, connection, area=area).order("population ASC, code5 ASC").limit(1).fetchone()
    if row is None:
        raise ValueError("No administrative area found")
    code5, name, population = row
    if type(population) is not int:
        raise TypeError("population must be an integer")
    return {"code5": code5, "name": name, "population": population}

## northernmost_station@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_stations


def northernmost_station(dataset_id):
    """Find the station with the largest latitude in the station dataset, with no filtering by operating state. Returns its name, latitude and longitude."""
    with connect_duckdb() as connection:
        row = load_stations(dataset_id, connection).order("latitude DESC, name ASC").limit(1).fetchone()
    if row is None:
        raise ValueError("No station found")
    name, latitude, longitude = row
    return {"name": name, "latitude": latitude, "longitude": longitude}

## route_summary@v1 (seed)
import json

from geo_voyager.control_primitives import call_service


def route_summary(origin_lat, origin_lon, destination_lat, destination_lon, costing="auto"):
    """Route between two points with Valhalla and return the length in kilometres and the time in seconds and minutes. costing is auto, pedestrian or bicycle."""
    request = {"locations": [{"lat": origin_lat, "lon": origin_lon}, {"lat": destination_lat, "lon": destination_lon}],
               "costing": costing, "units": "kilometers"}
    summary = json.loads(call_service("valhalla", path="/route", body=json.dumps(request),
                                      content_type="application/json"))["trip"]["summary"]
    return {"costing": costing, "length_km": summary["length"], "time_s": summary["time"], "time_min": summary["time"] / 60}

## tag_usage_count@v1 (seed)
import json

from geo_voyager.control_primitives import call_service


def tag_usage_count(key, value):
    """Count how many times the OSM tag key=value is used in the whole OSM database, with Taginfo's tag statistics. Returns the tag and the count."""
    payload = json.loads(call_service("taginfo", path="/api/4/tag/stats", params={"key": key, "value": value}))
    total = [item["count"] for item in payload["data"] if item["type"] == "all"]
    if len(total) != 1:
        raise ValueError(f"Taginfo returned no total for {key}={value}")
    return {"key": key, "value": value, "count": total[0]}

## top_areas_by_population@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def top_areas_by_population(dataset_id, count, area=None):
    """List the given number of administrative areas with the largest populations, largest first, in all areas or in one named set (for example 東京都23区). Each item has its 5-digit code, name and population."""
    with connect_duckdb() as connection:
        rows = load_admin_units(dataset_id, connection, area=area).order("population DESC, code5 ASC").limit(count).fetchall()
    if len(rows) != count:
        raise ValueError(f"{count} areas are required, {len(rows)} found")
    if any(type(population) is not int for _, _, population in rows):
        raise TypeError("population must be an integer")
    return [{"code5": code5, "name": name, "population": population} for code5, name, population in rows]

## top_cuisine_values_by_count@v1
from geo_voyager.control_primitives import call_service
import json


def top_cuisine_values_by_count(key="cuisine", limit=3):
    """実行時に指定されたキーの値の使用数を taginfo から取得し、使用数の多い順に上位3つの value と count を返す。"""
    payload = json.loads(call_service("taginfo", path="/api/4/key/values", params={"key": key, "page": "1", "rp": str(limit), "sortname": "count", "sortorder": "desc"}))
    data = payload.get("data")
    assert isinstance(data, list) and len(data) >= limit
    return [{"value": item["value"], "count": item["count"]} for item in data[:limit]]

## total_population@v1 (seed)
from geo_voyager.control_primitives import connect_duckdb, load_admin_units


def total_population(dataset_id, area=None):
    """Sum the population of the administrative areas, all of them or one named set (for example 東京都23区)."""
    with connect_duckdb() as connection:
        total = load_admin_units(dataset_id, connection, area=area).aggregate("sum(population) AS population_total").fetchone()[0]
    if type(total) is not int:
        raise TypeError("The population total must be an integer")
    return total
